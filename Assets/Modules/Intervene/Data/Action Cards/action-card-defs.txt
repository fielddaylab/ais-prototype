::Action-release-sterilized

@suit		research
@desc		-2 Invasive reproduction in a target tributary
@path		temp.png
@cost		1

    target: invasive, tributary
    verb: reproduce, -2, fixed
    specificity: specific

::Action-downstream-tributary-traps

::Action-pheromones
@title		Place Pheromones
@suit		tech
@desc		Traps on a target pathway catch 2x as many Invasives.
@path		temp.png
@cost		1

// trapboost takes a signed rate: positive catches more, negative catches fewer.
// "fixed" shifts the catch; "ratio" scales the trap's own strength (1.0 = catches twice as many).
// Applies to traps already on the pathway and to any placed there later.
@effect
    target: pathway
    verb: trapboost, 2, ratio
    specificity: specific
    maxtargets: 1
